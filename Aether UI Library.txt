--[[
  ÍCONES SEGUROS (estilo Rayfield Secure Mode)
  - Baixa o PNG uma vez e serve via getcustomasset (arquivo local)
  - Nunca usa rbxassetid:// de marketplace → nada para ContentProvider:GetAssetFetchStatus
  - Fonte: lucide.dev via latte-soft/lucide-roblox (compiled 256px)
]]
local ICON_FOLDER = "Aether_Icons"
local ICON_BASE = "https://raw.githubusercontent.com/latte-soft/lucide-roblox/master/icons/compiled/256px/"
local IconCache = {}

local function GetIcon(name)
    if not name or name == "" then return "" end
    name = tostring(name):lower():gsub("%s+", "-")
    if IconCache[name] then return IconCache[name] end

    local path = ICON_FOLDER .. "/" .. name .. ".png"
    pcall(function()
        if isfolder and not isfolder(ICON_FOLDER) then makefolder(ICON_FOLDER) end
    end)

    local asset = ""
    local hasFile = isfile and isfile(path)

    if not hasFile then
        local okGet, data = pcall(function()
            return game:HttpGet(ICON_BASE .. name .. ".png")
        end)
        if okGet and type(data) == "string" and #data > 100 then
            pcall(function()
                if writefile then writefile(path, data) end
            end)
            hasFile = true
        end
    end

    if hasFile and getcustomasset then
        local ok, a = pcall(getcustomasset, path)
        if ok and a and a ~= "" then asset = a end
    end

    IconCache[name] = asset
    return asset
end

-- SMART BOUNDS
local function SmartPos(pos, w, h)
    local vp = workspace.CurrentCamera.ViewportSize; local pad = 5
    local cx = math.clamp((pos.X.Scale*vp.X)+pos.X.Offset, w/2+pad, vp.X-w/2-pad)
    local cy = math.clamp((pos.Y.Scale*vp.Y)+pos.Y.Offset, h/2+pad, vp.Y-h/2-pad)
    return UDim2.new(0, cx, 0, cy)
end

-- DRAG
local function MakeDraggable(trigger, target, onEnd)
    local dragging, dragInput, dragStart, startPos
    trigger.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = i.Position; startPos = target.Position; LastInteract = tick()
            local con; con = i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false; con:Disconnect(); if onEnd then onEnd() end
                end
            end)
        end
    end)
    trigger.InputChanged:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseMovement
        or i.UserInputType == Enum.UserInputType.Touch then dragInput = i end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if i == dragInput and dragging then
            local d = i.Position-dragStart
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X, startPos.Y.Scale, startPos.Y.Offset+d.Y)
            LastInteract = tick()
        end
    end)
end

-- TOGGLE ANIMATION
local function ToggleUI(force)
    if isAnimating then return end
    if force ~= nil then IsOpen = force else IsOpen = not IsOpen end
    isAnimating = true; LastInteract = tick()
    if IsOpen then
        local safe = SmartPos(SavedPos, OpenW, OpenH)
        TitleLbl.Text = WinTitle .. " | " .. WinSub; Dot.BackgroundColor3 = C.Red
        Main.Position = safe
        SavedPos = safe
        Main:TweenSize(UDim2.new(0, OpenW, 0, OpenH), Enum.EasingDirection.Out, Enum.EasingStyle.Back, 0.5, true)
        Body.Position = UDim2.new(0, 0, 0, -100)
        TweenService:Create(Body, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            {Position = UDim2.new(0, 0, 0, 0), GroupTransparency = 0}):Play()
    else
        Dot.BackgroundColor3 = C.Green
        Tween(Body, 0.2, {GroupTransparency = 1}); Patch.Visible = false
        local safeClose = SmartPos(SavedPos, ClosedW, ClosedH)
        SavedPos = safeClose
        Main:TweenSize(UDim2.new(0, ClosedW, 0, ClosedH), Enum.EasingDirection.InOut, Enum.EasingStyle.Quad, 0.25, true)
        Main:TweenPosition(safeClose, Enum.EasingDirection.InOut, Enum.EasingStyle.Quad, 0.25, true)
        task.wait(0.25); TitleLbl.Text = WinTitle
        Main:TweenSize(UDim2.new(0, math.max(ClosedW - 30, 140), 0, ClosedH), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.1, true)
        task.wait(0.1)
        Main:TweenSize(UDim2.new(0, ClosedW, 0, ClosedH), Enum.EasingDirection.Out, Enum.EasingStyle.Elastic, 0.4, true)
    end
    task.wait(0.4); isAnimating = false
end

ToggleBtn.MouseButton1Click:Connect(function() ToggleUI() end)

    if gp then return end

    frame.Size = UDim2.new(0, 104, 0, 34); frame.LayoutOrder = TabCount

    local iconName = icon
    if type(name) == "table" then
        iconName = name.Icon or icon
        page = name.Page or page
        name = name.Name or name.Text or "Tab"
    end

    local hasIcon = iconName and iconName ~= ""
    local iconImg = nil

    if hasIcon then
        iconImg = Instance.new("ImageLabel", frame)
        iconImg.Name = "Icon"
        iconImg.BackgroundTransparency = 1
        iconImg.Size = UDim2.new(0, 14, 0, 14)
        iconImg.Position = UDim2.new(0, 8, 0.5, -7)
        iconImg.Image = GetIcon(iconName)
        iconImg.ImageColor3 = C.Dim
        iconImg.ImageTransparency = 0.15
        iconImg.ScaleType = Enum.ScaleType.Fit
        iconImg.ResampleMode = Enum.ResamplerMode.Pixelated
    end

    local lbl = Instance.new("TextLabel", frame)
    lbl.Name = "Lbl"; lbl.Text = tostring(name); lbl.Font = Enum.Font.GothamBold; lbl.TextSize = 12
    lbl.TextColor3 = C.Dim; lbl.BackgroundTransparency = 1
    if hasIcon then
        lbl.Size = UDim2.new(1, -30, 1, 0)
        lbl.Position = UDim2.new(0, 28, 0, 0)
    else
        lbl.Size = UDim2.new(1, -16, 1, 0)
        lbl.Position = UDim2.new(0, 10, 0, 0)
    end
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local hit = Instance.new("TextButton", frame)
    hit.BackgroundTransparency = 1; hit.Size = UDim2.new(1, 0, 1, 0); hit.Text = ""; hit.ZIndex = 5
    hit.MouseButton1Click:Connect(function()
        LastInteract = tick()
        for _, p in ipairs(Pages) do p.Visible = false end
        for _, b in ipairs(TabBtns) do
            Tween(b, 0.2, {BackgroundColor3 = C.Dark, BackgroundTransparency = 0.6})
            b.Lbl.TextColor3 = C.Dim
            if b:FindFirstChild("Icon") and b.Icon:IsA("ImageLabel") then
                Tween(b.Icon, 0.2, {ImageColor3 = C.Dim, ImageTransparency = 0.15})
            end

        page.Visible = true

        lbl.TextColor3 = C.White
        if iconImg then
            Tween(iconImg, 0.2, {ImageColor3 = C.White, ImageTransparency = 0})
        end

    table.insert(TabBtns, frame); return frame
end

    if type(opts) == "string" then opts = {Text = opts} end

local function Toggle(parent, opts)

    local function Set(val)
        state = val; Tween(pill, 0.2, {BackgroundColor3 = state and C.Green or C.Red})
        Tween(knob, 0.2, {Position = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)})
        if opts.Callback then opts.Callback(state) end
    end
    local hit = Instance.new("TextButton", bg)
    hit.BackgroundTransparency = 1; hit.Size = UDim2.new(1, 0, 1, 0)
    hit.Text = ""; hit.ZIndex = 5
    hit.MouseButton1Click:Connect(function() LastInteract = tick(); Set(not state) end)
    if hasKey then
        -- kb fica à esquerda do pill (pill está em X=1,-44; kb: 36px + 4px gap = X=1,-84)
        local kb = Instance.new("TextButton", bg)
        kb.Text = opts.Keybind and opts.Keybind.Name or "KEY"
        kb.Font = Enum.Font.GothamBold; kb.TextSize = 10

                if i.UserInputType == Enum.UserInputType.Keyboard then
                    kb.Text = i.KeyCode.Name; kb.TextColor3 = C.White

                    con:Disconnect()
                end
            end)
        end)
        if opts.Keybind then
            UserInputService.InputBegan:Connect(function(io, gp)
                if gp then return end
                if io.KeyCode == opts.Keybind then Set(not state) end
            end)
        end
    end
    DescLabel(wrap, opts.Description, 37)
    return {Set = Set, Get = function() return state end}
end

-- SLIDER — idêntico ao cateiro original
local function Slider(parent, opts)
    local min = opts.Min or 0
    local max = opts.Max or 100
    local cur = opts.Default or min
    local hasDesc = opts.Description and opts.Description ~= ""

    -- Base da source (conkerscateiro): track 25px, Corner 12
    local TRACK_H = 25
    local KNOB_W = 11
    local baseH = hasDesc and 60 or 45

    local SlContainer = Instance.new("Frame", parent)
    SlContainer.BackgroundTransparency = 1
    SlContainer.BorderSizePixel = 0
    SlContainer.Size = UDim2.new(1, -8, 0, baseH)

    local SlLabel = Instance.new("TextLabel", SlContainer)
    SlLabel.Text = opts.Text or "Slider"
    SlLabel.Size = UDim2.new(1, -50, 0, 15)
    SlLabel.BackgroundTransparency = 1
    SlLabel.TextColor3 = C.Dim
    SlLabel.Font = Enum.Font.GothamBold
    SlLabel.TextSize = 12
    SlLabel.TextXAlignment = Enum.TextXAlignment.Left

    -- número à direita do título (não fica atrás do knob)
    local SlVal = Instance.new("TextLabel", SlContainer)
    SlVal.Name = "Val"
    SlVal.BackgroundTransparency = 1
    SlVal.Size = UDim2.new(0, 44, 0, 15)
    SlVal.Position = UDim2.new(1, -44, 0, 0)
    SlVal.Text = tostring(cur)
    SlVal.TextColor3 = C.White
    SlVal.Font = Enum.Font.GothamBold
    SlVal.TextSize = 12
    SlVal.TextXAlignment = Enum.TextXAlignment.Right
    SlVal.ZIndex = 5

    -- track
    local SlBg = Instance.new("Frame", SlContainer)
    SlBg.BackgroundColor3 = C.Light
    SlBg.BackgroundTransparency = 0.3
    SlBg.BorderSizePixel = 0
    SlBg.Size = UDim2.new(1, 0, 0, TRACK_H)
    SlBg.Position = UDim2.new(0, 0, 0, 20)
    SlBg.ClipsDescendants = true
    Corner(SlBg, 12)
    Stroke(SlBg, C.Stroke, 1)

    local initRel = 0
    if max > min then
        initRel = math.clamp((cur - min) / (max - min), 0, 1)
    end

    --[[
      Fill arredondado SEM vazar:
      Roblox ClipsDescendants corta só no retângulo, NÃO no UICorner.
      Então o fill fica inset (2px) e com corner menor — fica redondo e
      nunca ultrapassa a borda visual do track.
    ]]
    local INSET = 2
    local SlFill = Instance.new("Frame", SlBg)
    SlFill.Name = "Fill"
    SlFill.BackgroundColor3 = C.Blue
    SlFill.BorderSizePixel = 0
    SlFill.Position = UDim2.new(0, INSET, 0, INSET)
    SlFill.Size = UDim2.new(initRel, -INSET * 2, 1, -INSET * 2)
    SlFill.ZIndex = 2
    Corner(SlFill, 10)

    -- knob (só ele controla)
    local SlKnob = Instance.new("TextButton", SlContainer)
    SlKnob.Name = "Knob"
    SlKnob.Text = ""
    SlKnob.AutoButtonColor = false
    SlKnob.BackgroundColor3 = C.White
    SlKnob.BorderSizePixel = 0
    SlKnob.Size = UDim2.new(0, KNOB_W, 0, TRACK_H + 4)
    SlKnob.AnchorPoint = Vector2.new(0.5, 0.5)
    SlKnob.Position = UDim2.new(initRel, -3, 0, 20 + TRACK_H / 2)
    SlKnob.ZIndex = 10
    Corner(SlKnob, 100)
    Stroke(SlKnob, Color3.new(0, 0, 0), 1)

    local dragging = false
    local function applyVisual(rel)
        rel = math.clamp(rel, 0, 1)
        -- largura do fill: scale * (track - insets). Em UDim2, offset negativo
        -- com scale funciona: Size = (rel, -INSET*2) a partir de Position X = INSET
        SlFill.Size = UDim2.new(rel, -INSET * 2, 1, -INSET * 2)
        -- quando rel é muito pequeno, evita width negativo visual
        if rel <= 0 then
            SlFill.Size = UDim2.new(0, 0, 1, -INSET * 2)
        end
        SlKnob.Position = UDim2.new(rel, -3, 0, 20 + TRACK_H / 2)
        SlVal.Text = tostring(cur)
    end

    local function upd(ax)
        local absX = SlBg.AbsolutePosition.X
        local absW = math.max(SlBg.AbsoluteSize.X, 1)
        local rel = math.clamp((ax - absX) / absW, 0, 1)
        cur = math.floor(min + rel * (max - min) + 0.5)
        cur = math.clamp(cur, min, max)
        rel = (max > min) and ((cur - min) / (max - min)) or 0
        applyVisual(rel)
        if opts.Callback then opts.Callback(cur) end
    end

    SlKnob.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
            if IsInteracting then return end
            dragging = true
            IsInteracting = true
            LastInteract = tick()
            local scroll = GetScroll(SlContainer)
            if scroll then scroll.ScrollingEnabled = false end
            upd(i.Position.X)
            local endCon
            endCon = UserInputService.InputEnded:Connect(function(ie)
                if ie.UserInputType == Enum.UserInputType.MouseButton1
                    or ie.UserInputType == Enum.UserInputType.Touch then
                    dragging = false
                    IsInteracting = false
                    endCon:Disconnect()
                    if scroll then scroll.ScrollingEnabled = true end
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
            upd(i.Position.X)
            LastInteract = tick()
        end
    end)

    if hasDesc then
        local d = Instance.new("TextLabel", SlContainer)
        d.Text = opts.Description
        d.Font = Enum.Font.Gotham
        d.TextSize = 10
        d.TextColor3 = C.Dim
        d.BackgroundTransparency = 1
        d.Size = UDim2.new(1, 0, 0, 14)
        d.Position = UDim2.new(0, 2, 0, 48)
        d.TextXAlignment = Enum.TextXAlignment.Left
        d.TextTransparency = 0.3
    end

    applyVisual(initRel)

    return {
        Set = function(v)
            cur = math.clamp(v, min, max)
            local rel = (max > min) and ((cur - min) / (max - min)) or 0
            applyVisual(rel)
        end,
        Get = function() return cur end,
    }
end

    local isOpen = false
    local BASE_H = 34

        return (opts.Text and (opts.Text .. ":  ") or "") .. current
    end

    local wrap = Instance.new("Frame", parent)

    arrow.Size = UDim2.new(0, 14, 0, 14)

    arrow.ImageTransparency = 0.1
    arrow.ScaleType = Enum.ScaleType.Fit
    arrow.ResampleMode = Enum.ResamplerMode.Pixelated
    arrow.Rotation = 0

    if hasDesc then
        desc = Instance.new("TextLabel", wrap)

        desc.TextColor3 = C.Dim; desc.BackgroundTransparency = 1
        desc.Size = UDim2.new(1, -4, 0, 14); desc.Position = UDim2.new(0, 2, 0, BASE_H + 2)
        desc.TextXAlignment = Enum.TextXAlignment.Left; desc.TextTransparency = 0.3

    sep.BackgroundColor3 = C.Stroke; sep.BackgroundTransparency = 0.5; sep.BorderSizePixel = 0

    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    Pad(listFrame, 4, 4, 4, 4)

    local function getOpenH()
        return BASE_H + 8 + math.min(#items, 5) * (ITEM_H + 4) + 8
    end

    local function close()
        isOpen = false
        Tween(arrow, 0.2, {Rotation = 0})
        Tween(box, 0.3, {Size = UDim2.new(1, 0, 0, BASE_H)})
        Tween(wrap, 0.3, {Size = UDim2.new(1, -8, 0, BASE_H + (hasDesc and 16 or 0))})

        end
    end

    local function rebuild()
        for _, c in ipairs(listFrame:GetChildren()) do

            local oh = getOpenH()

            Tween(wrap, 0.35, {Size = UDim2.new(1, -8, 0, oh + (hasDesc and 16 or 0))})

                Tween(desc, 0.35, {Position = UDim2.new(0, 2, 0, oh + 2)})
            end
        else close() end
    end)

    return {
        Set = function(v2) current = v2; lbl.Text = labelText() end,
        Get = function() return current end,
        SetItems = function(t) items = t; if isOpen then rebuild() end end,
    }
end

            local r = tonumber(hex:sub(1, 2), 16); local g = tonumber(hex:sub(3, 4), 16); local b = tonumber(hex:sub(5, 6), 16)

                v = 1-math.clamp((inp.Position.Y-svHit.AbsolutePosition.Y)/math.max(svHit.AbsoluteSize.Y, 1), 0, 1)
                apply()
            end

            local move; move = UserInputService.InputChanged:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then upd(inp) end
            end)
            local endC; endC = UserInputService.InputEnded:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
                    IsInteracting = false; move:Disconnect(); endC:Disconnect()
                    if scroll then scroll.ScrollingEnabled = true end
                end
            end)
        end
    end)

    -- Drag Hue
    hHit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            if IsInteracting then return end
            IsInteracting = true; LastInteract = tick()
            local scroll = GetScroll(box); if scroll then scroll.ScrollingEnabled = false end
            local function upd(inp)
                h = math.clamp((inp.Position.Y-hHit.AbsolutePosition.Y)/math.max(hHit.AbsoluteSize.Y, 1), 0, 1)
                apply()
            end
            upd(i)
            local move; move = UserInputService.InputChanged:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then upd(inp) end
            end)

                if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
                    IsInteracting = false; move:Disconnect(); endC:Disconnect()
                    if scroll then scroll.ScrollingEnabled = true end
                end
            end)
        end
    end)

    return {
        Get = function() return color end,
        Set = function(col) color = col; h, s, v = Color3.toHSV(col); apply() end,
    }
end

-- KEYBIND
local function Keybind(parent, opts)
    opts = opts or {}
    local current = opts.Default or Enum.KeyCode.Unknown
    local listening = false
    local wrap = Wrap(parent, 34)
    local bg = Instance.new("Frame", wrap)
    bg.BackgroundColor3 = C.Light; bg.BackgroundTransparency = 0.4
    bg.BorderSizePixel = 0; bg.Size = UDim2.new(1, 0, 0, 34)
    Corner(bg, 10); Stroke(bg, C.Stroke, 1)

    local lbl = Instance.new("TextLabel", bg)
    lbl.Text = opts.Text or "Keybind"; lbl.Font = Enum.Font.GothamBold; lbl.TextSize = 13
    lbl.TextColor3 = C.White; lbl.BackgroundTransparency = 1
    lbl.Size = UDim2.new(1, -90, 1, 0); lbl.Position = UDim2.new(0, 10, 0, 0)
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local kb = Instance.new("TextButton", bg)
    kb.Text = (current and current.Name) or "NONE"
    kb.Font = Enum.Font.GothamBold; kb.TextSize = 11
    kb.BackgroundColor3 = C.Dark; kb.BackgroundTransparency = 0.4; kb.TextColor3 = C.Dim
    kb.Size = UDim2.new(0, 72, 0, 22); kb.Position = UDim2.new(1, -80, 0.5, -11)
    kb.AutoButtonColor = false; Corner(kb, 8); Stroke(kb, C.Stroke, 1)

    kb.MouseButton1Click:Connect(function()
        if IsInteracting then return end
        LastInteract = tick()
        listening = true
        kb.Text = "..."; kb.TextColor3 = C.Green
    end)

    UserInputService.InputBegan:Connect(function(i, gp)
        if listening then
            if i.UserInputType == Enum.UserInputType.Keyboard then
                current = i.KeyCode
                kb.Text = current.Name; kb.TextColor3 = C.White
                listening = false
                if opts.Callback then opts.Callback(current) end
            end
        elseif not gp and current and i.KeyCode == current then
            if opts.Pressed then opts.Pressed() end
        end
    end)

    DescLabel(wrap, opts.Description, 37)
    return {
        Set = function(k) current = k; kb.Text = k and k.Name or "NONE" end,
        Get = function() return current end,
    }
end

-- API PÚBLICA
--   Library.CreateWindow({ Title, Subtitle, Width, Height, ToggleKey }) → Window
--   Window:CreateTab({ Name, Icon }) → Tab  -- Icon: nome Lucide (ex: "home", "settings")
--   Tab:AddSection / AddLabel / AddParagraph / AddButton / AddToggle
--   Tab:AddSlider / AddDropdown / AddMultiDropdown / AddColorPicker
--   Tab:AddTextBox / AddKeybind / AddDialog
--   Window:Notify / Window:Toggle / Window:Dialog

local function CreateTabAPI(page)
    local Tab = { Page = page }

    function Tab:AddSection(opts)
        if type(opts) == "string" then opts = { Text = opts } end
        return Section(page, opts or {})
    end
    function Tab:AddLabel(opts)
        if type(opts) == "string" then opts = { Text = opts } end
        return Label(page, opts or {})
    end
    function Tab:AddParagraph(opts)
        return Paragraph(page, opts or {})
    end
    function Tab:AddButton(opts)
        return Button(page, opts or {})
    end
    function Tab:AddToggle(opts)
        return Toggle(page, opts or {})
    end
    function Tab:AddSlider(opts)
        return Slider(page, opts or {})
    end
    function Tab:AddDropdown(opts)
        return Dropdown(page, opts or {})
    end
    function Tab:AddMultiDropdown(opts)
        return MultiDropdown(page, opts or {})
    end
    function Tab:AddColorPicker(opts)
        return ColorPicker(page, opts or {})
    end
    function Tab:AddTextBox(opts)
        return TextBox(page, opts or {})
    end
    function Tab:AddKeybind(opts)
        return Keybind(page, opts or {})
    end
    function Tab:AddDialog(opts)
        return Dialog(page, opts or {})
    end

    return Tab
end

local function CreateWindow(config)
    if type(config) ~= "table" then config = {} end

    if config.Title then
        WinTitle = tostring(config.Title)
        TitleLbl.Text = WinTitle
    end
    if config.Subtitle then
        WinSub = tostring(config.Subtitle)
    end
    if config.Width then
        OpenW = math.max(tonumber(config.Width) or OpenW, 280)
    end
    if config.Height then
        OpenH = math.max(tonumber(config.Height) or OpenH, 180)
    end
    if config.ToggleKey then
        UserInputService.InputBegan:Connect(function(io, gp)
            if gp then return end
            if io.KeyCode == config.ToggleKey then
                ToggleUI()
            end
        end)
    end

    local Window = {
        Tabs = {},
        FirstTab = nil,
    }

    function Window:CreateTab(opts)
        if type(opts) == "string" then
            opts = { Name = opts }
        end
        opts = opts or {}
        local name = opts.Name or opts.Text or "Tab"
        local icon = opts.Icon
        local page = NewPage()
        local tabFrame = NewTab(name, page, icon)
        local tabAPI = CreateTabAPI(page)
        tabAPI.Frame = tabFrame
        tabAPI.Name = name

        if not Window.FirstTab then
            Window.FirstTab = tabFrame
            SetFirstTab(tabFrame)
            page.Visible = true
        end

        table.insert(Window.Tabs, tabAPI)
        return tabAPI
    end

    function Window:Notify(msg, dur, col)
        return Notify(msg, dur, col)
    end

    function Window:Toggle(force)
        return ToggleUI(force)
    end

    function Window:Dialog(opts)
        return ShowDialog(opts or {})
    end

    return Window
end

return {
    CreateWindow = CreateWindow,
}
